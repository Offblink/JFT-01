📄 1. UserMapperAnnotation.java（注解方式完整示例）

package com.example.mapper;

import com.example.entity.User;
import org.apache.ibatis.annotations.*;

import java.util.List;
import java.util.Map;

/**
 * 用户 Mapper 接口（注解方式）
 *
 * 所有 SQL 语句都写在注解中，不需要 XML 文件
 */
public interface UserMapperAnnotation {

    // ==================== 基础 CRUD ====================

    /**
     * 查询所有用户
     * @return 用户列表
     */
    @Select("SELECT * FROM user")
    List findAll();

    /**
     * 根据 ID 查询用户
     * @param id 用户ID
     * @return 用户对象
     */
    @Select("SELECT * FROM user WHERE id = #{id}")
    User findById(Integer id);

    /**
     * 添加用户（获取自增主键）
     * @param user 用户对象
     * @return 影响行数
     */
    @Insert("INSERT INTO user(username, password, email) VALUES(#{username}, #{password}, #{email})")
    @Options(useGeneratedKeys = true, keyProperty = "id")
    int addUser(User user);

    /**
     * 更新用户
     * @param user 用户对象
     * @return 影响行数
     */
    @Update("UPDATE user SET username=#{username}, password=#{password}, email=#{email} WHERE id=#{id}")
    int updateUser(User user);

    /**
     * 删除用户
     * @param id 用户ID
     * @return 影响行数
     */
    @Delete("DELETE FROM user WHERE id = #{id}")
    int deleteUser(Integer id);

    // ==================== 多参数传递（@Param 注解） ====================

    /**
     * 多条件查询（方式1：@Param 注解）
     * @param username 用户名
     * @param email 邮箱
     * @return 用户列表
     */
    @Select("SELECT * FROM user WHERE username=#{username} AND email=#{email}")
    List findByNameAndEmail(@Param("username") String username, @Param("email") String email);

    /**
     * 多条件查询（方式2：Map 集合）
     * @param map 包含查询条件的 Map
     * @return 用户列表
     */
    @Select("SELECT * FROM user WHERE username=#{username} AND email=#{email}")
    List findUserByMap(Map map);

    /**
     * 多条件查询（方式3：实体类）
     * @param user 包含查询条件的用户对象
     * @return 用户列表
     */
    @Select("SELECT * FROM user WHERE username=#{username} AND email=#{email}")
    List findUserByEntity(User user);

    // ==================== ${} 用法（慎用） ====================

    /**
     * 模糊查询（${} 用法，存在 SQL 注入风险）
     * @param username 用户名（支持模糊匹配）
     * @return 用户列表
     */
    @Select("SELECT * FROM user WHERE username LIKE '%${username}%'")
    List findByNameLike(@Param("username") String username);

    /**
     * 动态表名查询（${} 正确用法）
     * @param tableName 表名
     * @return 用户列表
     */
    @Select("SELECT * FROM ${tableName}")
    List findAllByTableName(@Param("tableName") String tableName);

    /**
     * 动态排序（${} 正确用法）
     * @param column 排序字段
     * @param order 排序方式（ASC/DESC）
     * @return 用户列表
     */
    @Select("SELECT * FROM user ORDER BY ${column} ${order}")
    List findByOrder(@Param("column") String column, @Param("order") String order);

    // ==================== 用户登录（SQL 注入演示） ====================

    /**
     * 用户登录（不安全：使用 ${}）
     * @param username 用户名
     * @param password 密码
     * @return 用户对象
     */
    @Select("SELECT * FROM user WHERE username = '${username}' AND password = '${password}'")
    User loginUnsafe(@Param("username") String username, @Param("password") String password);

    /**
     * 用户登录（安全：使用 #{}）
     * @param username 用户名
     * @param password 密码
     * @return 用户对象
     */
    @Select("SELECT * FROM user WHERE username = #{username} AND password = #{password}")
    User loginSafe(@Param("username") String username, @Param("password") String password);
}

📄 2. UserMapperAnnotationTest.java（测试类完整示例）

package com.example;

import com.example.entity.User;
import com.example.mapper.UserMapperAnnotation;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import org.junit.After;
import org.junit.Before;
import org.junit.Test;

import java.io.IOException;
import java.io.InputStream;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * 注解方式 Mapper 测试类
 *
 * 测试 UserMapperAnnotation 接口的所有方法
 */
public class UserMapperAnnotationTest {

    private InputStream is;
    private SqlSession session;
    private UserMapperAnnotation userMapper;

    /**
     * 初始化方法
     */
    @Before
    public void init() throws IOException {
        System.out.println("========== 初始化开始 ==========");
        is = Resources.getResourceAsStream("mybatis-config.xml");
        SqlSessionFactory factory = new SqlSessionFactoryBuilder().build(is);
        session = factory.openSession(true);
        userMapper = session.getMapper(UserMapperAnnotation.class);
        System.out.println("========== 初始化完成 ==========");
    }

    /**
     * 销毁方法
     */
    @After
    public void destroy() throws IOException {
        System.out.println("========== 销毁开始 ==========");
        if (session != null) session.close();
        if (is != null) is.close();
        System.out.println("========== 销毁完成 ==========");
    }

    // ==================== 测试基础 CRUD ====================

    @Test
    public void testFindAll() {
        System.out.println("========== 测试查询所有用户 ==========");
        List users = userMapper.findAll();
        for (User user : users) {
            System.out.println(user);
        }
        assert users.size() > 0;
    }

    @Test
    public void testFindById() {
        System.out.println("========== 测试根据 ID 查询用户 ==========");
        User user = userMapper.findById(1);
        System.out.println(user);
        assert user != null;
        assert user.getId() == 1;
    }

    @Test
    public void testAddUser() {
        System.out.println("========== 测试添加用户 ==========");
        User user = new User();
        user.setUsername("注解用户");
        user.setPassword("123456");
        user.setEmail("annotation@qq.com");

        int rows = userMapper.addUser(user);
        System.out.println("影响行数：" + rows);
        System.out.println("自增主键：" + user.getId());  // 获取自增主键
        assert rows > 0;
    }

    // ==================== 测试多参数传递 ====================

    @Test
    public void testFindByNameAndEmail() {
        System.out.println("========== 测试多参数传递（@Param） ==========");
        List users = userMapper.findByNameAndEmail("张三", "zhangsan@qq.com");
        for (User user : users) {
            System.out.println(user);
        }
    }

    @Test
    public void testFindUserByMap() {
        System.out.println("========== 测试多参数传递（Map） ==========");
        Map map = new HashMap<>();
        map.put("username", "张三");
        map.put("email", "zhangsan@qq.com");

        List users = userMapper.findUserByMap(map);
        for (User user : users) {
            System.out.println(user);
        }
    }

    @Test
    public void testFindUserByEntity() {
        System.out.println("========== 测试多参数传递（实体类） ==========");
        User user = new User();
        user.setUsername("张三");
        user.setEmail("zhangsan@qq.com");

        List users = userMapper.findUserByEntity(user);
        for (User user2 : users) {
            System.out.println(user2);
        }
    }

    // ==================== 测试 ${} 用法 ====================

    @Test
    public void testFindByNameLike() {
        System.out.println("========== 测试模糊查询（${}） ==========");
        // 注意：这里存在 SQL 注入风险（演示用）
        List users = userMapper.findByNameLike("张");
        for (User user : users) {
            System.out.println(user);
        }
    }

    @Test
    public void testFindAllByTableName() {
        System.out.println("========== 测试动态表名（${}） ==========");
        List users = userMapper.findAllByTableName("user");
        for (User user : users) {
            System.out.println(user);
        }
    }

    @Test
    public void testFindByOrder() {
        System.out.println("========== 测试动态排序（${}） ==========");
        List users = userMapper.findByOrder("id", "DESC");
        for (User user : users) {
            System.out.println(user);
        }
    }

    // ==================== 测试 SQL 注入 ====================

    @Test
    public void testSqlInjectionUnsafe() {
        System.out.println("========== 测试 SQL 注入（不安全） ==========");
        
        // 正常登录
        User user1 = userMapper.loginUnsafe("张三", "123456");
        System.out.println("正常登录：" + user1);
        
        // SQL 注入攻击
        System.out.println("开始 SQL 注入攻击...");
        User user2 = userMapper.loginUnsafe("admin' -- ", "任意密码");
        System.out.println("SQL 注入登录：" + user2);
        
        if (user2 != null) {
            System.out.println("⚠️ 警告：SQL 注入攻击成功！");
        } else {
            System.out.println("✅ SQL 注入攻击失败（正常）");
        }
    }

    @Test
    public void testSqlInjectionSafe() {
        System.out.println("========== 测试 SQL 注入（安全） ==========");
        
        // 正常登录
        User user1 = userMapper.loginSafe("张三", "123456");
        System.out.println("正常登录：" + user1);
        
        // SQL 注入攻击（失败）
        System.out.println("开始 SQL 注入攻击...");
        User user2 = userMapper.loginSafe("admin' -- ", "任意密码");
        System.out.println("SQL 注入登录：" + user2);
        
        if (user2 != null) {
            System.out.println("⚠️ 警告：SQL 注入攻击成功！");
        } else {
            System.out.println("✅ SQL 注入攻击失败（防御成功）");
        }
    }
}

📝 常见问题与解决方案

问题 原因 解决方案
Parameter 'XXX' not found 多参数未使用 @Param 注解 在方法参数前添加 @Param("参数名") 注解
SQL 注入攻击成功 使用 ${} 拼接用户输入 改用 #{} 占位符；表名/列名用 ${} 时需校验输入
注解方式不支持动态 SQL 注解方式不适合复杂 SQL 改用 XML 方式，或使用 <script> 标签（不推荐）
自增主键获取失败 未配置 @Options 注解 添加 @Options(useGeneratedKeys = true, keyProperty = "id")
