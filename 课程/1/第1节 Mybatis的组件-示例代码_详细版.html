📄 1. 数据库初始化脚本（mybatis_db.sql）

-- 创建数据库
CREATE DATABASE IF NOT EXISTS mybatis_db DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;

-- 使用数据库
USE mybatis_db;

-- 删除表（如果存在）
DROP TABLE IF EXISTS user;

-- 创建用户表
CREATE TABLE user (
    id INT(11) PRIMARY KEY AUTO_INCREMENT COMMENT '用户ID',
    username VARCHAR(50) NOT NULL UNIQUE COMMENT '用户名',
    password VARCHAR(50) NOT NULL COMMENT '密码',
    email VARCHAR(100) COMMENT '邮箱',
    create_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    update_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 插入测试数据
INSERT INTO user (username, password, email) VALUES
('张三', '123456', 'zhangsan@qq.com'),
('李四', 'abcdef', 'lisi@qq.com'),
('王五', '111111', 'wangwu@qq.com'),
('赵六', '222222', 'zhaoliu@qq.com'),
('钱七', '333333', 'qianqi@qq.com');

-- 查询验证
SELECT * FROM user;

执行步骤：

  1. 打开 MySQL 客户端（如 Navicat、SQLyog、或 CMD）
  2. 复制上述 SQL 脚本
  3. 执行脚本，创建数据库和表
  4. 验证数据：SELECT * FROM mybatis_db.user;

📄 3. 实体类（User.java）

package com.example.entity;

import java.io.Serializable;
import java.util.Date;

/**
 * 用户实体类
 * 对应数据库表：user
 *
 *  implements Serializable：实现序列化接口（可选，但推荐）
 *  用于分布式环境、缓存、网络传输等场景
 */
public class User implements Serializable {

    // 字段（对应数据库表的列）
    private Integer id;       // 用户ID（对应 user.id）
    private String username;  // 用户名（对应 user.username）
    private String password;  // 密码（对应 user.password）
    private String email;     // 邮箱（对应 user.email）
    private Date createTime;  // 创建时间（对应 user.create_time）
    private Date updateTime;  // 更新时间（对应 user.update_time）

    // 构造方法（必须有无参构造，Mybatis 反射创建对象需要）
    public User() {
    }

    public User(String username, String password, String email) {
        this.username = username;
        this.password = password;
        this.email = email;
    }

    // getter 和 setter 方法（必须有，否则 Mybatis 无法封装数据）
    public Integer getId() {
        return id;
    }

    public void setId(Integer id) {
        this.id = id;
    }

    public String getUsername() {
        return username;
    }

    public void setUsername(String username) {
        this.username = username;
    }

    public String getPassword() {
        return password;
    }

    public void setPassword(String password) {
        this.password = password;
    }

    public String getEmail() {
        return email;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public Date getCreateTime() {
        return createTime;
    }

    public void setCreateTime(Date createTime) {
        this.createTime = createTime;
    }

    public Date getUpdateTime() {
        return updateTime;
    }

    public void setUpdateTime(Date updateTime) {
        this.updateTime = updateTime;
    }

    // toString 方法（方便调试和日志输出）
    @Override
    public String toString() {
        return "User{" +
                "id=" + id +
                ", username='" + username + '\'' +
                ", password='" + password + '\'' +
                ", email='" + email + '\'' +
                ", createTime=" + createTime +
                ", updateTime=" + updateTime +
                '}';
    }
}

实体类编写规范：

  • ✅ 实现 Serializable 接口（推荐）
  • ✅ 提供无参构造方法（必须）
  • ✅ 提供 getter 和 setter 方法（必须）
  • ✅ 重写 toString() 方法（推荐，方便调试）
  • ✅ 字段名与数据库列名一致（或配置驼峰命名自动映射）

📄 4. Mapper 接口（UserMapper.java）

package com.example.mapper;

import com.example.entity.User;
import java.util.List;

/**
 * 用户 Mapper 接口
 *
 * 对应 Mapper XML 文件：UserMapper.xml
 * Mybatis 会自动生成该接口的代理对象
 */
public interface UserMapper {

    /**
     * 查询所有用户
     * @return 用户列表
     *
     * 对应 UserMapper.xml 中的：
     * <select id="findAll" resultType="com.example.entity.User">
     */
    List findAll();

    /**
     * 根据 ID 查询用户
     * @param id 用户ID
     * @return 用户对象
     *
     * 对应 UserMapper.xml 中的：
     * <select id="findById" parameterType="int" resultType="com.example.entity.User">
     */
    User findById(Integer id);

    /**
     * 添加用户
     * @param user 用户对象
     * @return 影响行数
     *
     * 对应 UserMapper.xml 中的：
     * <insert id="addUser" parameterType="com.example.entity.User">
     */
    int addUser(User user);

    /**
     * 更新用户
     * @param user 用户对象
     * @return 影响行数
     *
     * 对应 UserMapper.xml 中的：
     * <update id="updateUser" parameterType="com.example.entity.User">
     */
    int updateUser(User user);

    /**
     * 删除用户
     * @param id 用户ID
     * @return 影响行数
     *
     * 对应 UserMapper.xml 中的：
     * <delete id="deleteUser" parameterType="int">
     */
    int deleteUser(Integer id);
}

Mapper 接口编写规范：

  • ✅ 接口方法名必须与 Mapper XML 中的 SQL 语句 id 一致
  • ✅ 返回值类型必须与 Mapper XML 中的 resultType 一致
  • ✅ 参数类型必须与 Mapper XML 中的 parameterType 一致
  • ✅ 包名 + 接口名 = Mapper XML 中的 namespace

Mapper XML 编写规范：

  • ✅ namespace 必须等于 Mapper 接口的全限定名
  • ✅ SQL 语句的 id 必须等于 Mapper 接口的方法名
  • ✅ parameterType 和 resultType 必须正确配置
  • ✅ SQL 语句中的参数使用 #{...} 占位符
  • ✅ 数据库列名与实体类字段名不一致时，使用别名（AS）或 ResultMap 映射

📄 6. 数据库配置（db.properties）

# 数据库驱动（MySQL 8.x 使用 com.mysql.cj.jdbc.Driver）
jdbc.driver=com.mysql.cj.jdbc.Driver

# 数据库连接 URL
# useSSL=false：不使用 SSL 加密（开发环境）
# serverTimezone=UTC：设置时区（解决时区错误）
# characterEncoding=utf8：设置字符编码
# useUnicode=true：启用 Unicode 编码
jdbc.url=jdbc:mysql://localhost:3306/mybatis_db?useSSL=false&serverTimezone=UTC&characterEncoding=utf8&useUnicode=true

# 数据库用户名（修改为你的用户名）
jdbc.username=root

# 数据库密码（修改为你的密码）
jdbc.password=123456

配置说明：

参数 说明
useSSL 是否使用 SSL 加密（开发环境设为 false）
serverTimezone 服务器时区（解决时区错误，推荐 UTC 或 Asia/Shanghai）
characterEncoding 字符编码（推荐使用 utf8 或 utf8mb4）
useUnicode 是否启用 Unicode 编码（推荐 true）

📄 7. Mybatis 全局配置（mybatis-config.xml）

配置说明：

  • <properties>：引入外部配置文件（db.properties）
  • <settings>：全局参数设置
    • mapUnderscoreToCamelCase：驼峰命名自动映射
    • logImpl：日志实现（STDOUT_LOGGING：控制台打印）
    • cacheEnabled：二级缓存开关
  • <environments>：环境配置（可以配置多个环境）
  • <mappers>：注册 Mapper XML 文件

📄 8. 测试类（MybatisTest.java）

package com.example;

import com.example.entity.User;
import com.example.mapper.UserMapper;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import org.junit.After;
import org.junit.Before;
import org.junit.Test;

import java.io.IOException;
import java.io.InputStream;
import java.util.Date;
import java.util.List;

/**
 * Mybatis 测试类
 *
 * 使用 @Before 和 @After 注解优化代码
 */
public class MybatisTest {

    private InputStream is;
    private SqlSessionFactory factory;
    private SqlSession session;
    private UserMapper userMapper;

    /**
     * 初始化方法（在每个测试方法执行前执行）
     */
    @Before
    public void init() throws IOException {
        // 1. 加载配置文件
        is = Resources.getResourceAsStream("mybatis-config.xml");

        // 2. 创建 SqlSessionFactory
        factory = new SqlSessionFactoryBuilder().build(is);

        // 3. 创建 SqlSession
        session = factory.openSession(true);  // true：自动提交事务

        // 4. 获取 Mapper 代理对象
        userMapper = session.getMapper(UserMapper.class);
    }

    /**
     * 销毁方法（在每个测试方法执行后执行）
     */
    @After
    public void destroy() throws IOException {
        // 关闭资源
        if (session != null) {
            session.close();
        }
        if (is != null) {
            is.close();
        }
    }

    /**
     * 测试查询所有用户
     */
    @Test
    public void testFindAll() {
        // 执行查询
        List users = userMapper.findAll();

        // 遍历结果
        for (User user : users) {
            System.out.println(user);
        }

        // 断言（可选）
        assert users.size() > 0 : "用户列表不能为空";
    }

    /**
     * 测试根据 ID 查询用户
     */
    @Test
    public void testFindById() {
        // 执行查询
        User user = userMapper.findById(1);

        // 打印结果
        System.out.println(user);

        // 断言
        assert user != null : "用户不能为空";
        assert user.getId() == 1 : "用户ID必须是1";
    }

    /**
     * 测试添加用户
     */
    @Test
    public void testAddUser() {
        // 创建用户对象
        User user = new User();
        user.setUsername("测试用户");
        user.setPassword("123456");
        user.setEmail("test@qq.com");

        // 执行添加
        int rows = userMapper.addUser(user);

        // 打印结果
        System.out.println("影响行数：" + rows);

        // 断言
        assert rows > 0 : "添加失败";
    }

    /**
     * 测试更新用户
     */
    @Test
    public void testUpdateUser() {
        // 查询用户
        User user = userMapper.findById(1);

        // 修改信息
        user.setUsername("张三更新");
        user.setEmail("zhangsan_new@qq.com");

        // 执行更新
        int rows = userMapper.updateUser(user);

        // 打印结果
        System.out.println("影响行数：" + rows);

        // 断言
        assert rows > 0 : "更新失败";
    }

    /**
     * 测试删除用户
     */
    @Test
    public void testDeleteUser() {
        // 执行删除
        int rows = userMapper.deleteUser(6);

        // 打印结果
        System.out.println("影响行数：" + rows);

        // 断言
        assert rows > 0 : "删除失败";
    }
}

测试类编写规范：

  • ✅ 使用 @Before 注解初始化资源（每个测试方法执行前执行）
  • ✅ 使用 @After 注解关闭资源（每个测试方法执行后执行）
  • ✅ 使用 factory.openSession(true) 开启自动提交事务
  • ✅ 使用断言（assert）验证测试结果

📝 10. 常见问题与解决方案

问题 原因 解决方案
ClassNotFoundException: com.mysql.cj.jdbc.Driver MySQL 驱动依赖未下载或版本错误 检查 pom.xml，确认 mysql-connector-java 依赖已添加，手动执行 mvn clean install
SQLException: The server time zone value 'XXX' is unrecognized 时区错误 在 db.properties 的 jdbc.url 中添加 serverTimezone=UTC 或 serverTimezone=Asia/Shanghai
BindingException: Invalid bound statement: com.example.mapper.UserMapper.findAll Mapper 接口与 XML 不匹配 检查 namespace、id 是否一致；检查 Mapper XML 是否已注册到 mybatis-config.xml
NullPointerException SqlSession 未开启自动提交，且未手动 commit 使用 factory.openSession(true) 开启自动提交，或在增删改操作后手动 session.commit()
SQL 语句打印乱码 字符编码问题 在 db.properties 中添加 characterEncoding=utf8&useUnicode=true
