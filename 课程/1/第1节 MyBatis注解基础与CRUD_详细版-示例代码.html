5. Mapper接口

src/main/java/com/example/mapper/UserMapper.java
package com.example.mapper;

import com.example.pojo.User;
import org.apache.ibatis.annotations.*;
import java.util.List;
import java.util.Map;

/**
 * 用户Mapper接口
 * 使用注解配置SQL
 */
public interface UserMapper {
    
    // ==================== 查询操作 ====================
    
    /**
     * 查询所有未删除的用户
     */
    @Select("SELECT * FROM user WHERE deleted = 0")
    List<User> findAll();
    
    /**
     * 根据ID查询用户
     */
    @Select("SELECT * FROM user WHERE id = #{id} AND deleted = 0")
    User findById(Integer id);
    
    /**
     * 根据用户名查询用户
     */
    @Select("SELECT * FROM user WHERE username = #{username} AND deleted = 0")
    User findByUsername(String username);
    
    /**
     * 根据用户名模糊查询
     */
    @Select("SELECT * FROM user WHERE username LIKE CONCAT('%', #{keyword}, '%') AND deleted = 0")
    List<User> findByKeyword(String keyword);
    
    /**
     * 多参数查询（使用@Param注解）
     */
    @Select("SELECT * FROM user WHERE username = #{name} AND age >= #{minAge} AND deleted = 0")
    List<User> findByUsernameAndAge(@Param("name") String username, 
                                     @Param("minAge") Integer age);
    
    /**
     * 动态查询（使用<script>标签）
     */
    @Select({
        "<script>",
        "SELECT * FROM user WHERE deleted = 0",
        "<if test='username != null and username != \"\"'>",
        "AND username LIKE CONCAT('%', #{username}, '%')",
        "</if>",
        "<if test='minAge != null'>",
        "AND age >= #{minAge}",
        "</if>",
        "<if test='maxAge != null'>",
        "AND age <= #{maxAge}",
        "</if>",
        "<if test='email != null and email != \"\"'>",
        "AND email = #{email}",
        "</if>",
        "ORDER BY id DESC",
        "</script>"
    })
    List<User> findByCondition(@Param("username") String username,
                                @Param("minAge") Integer minAge,
                                @Param("maxAge") Integer maxAge,
                                @Param("email") String email);
    
    /**
     * 动态查询（使用@SelectProvider）
     */
    @SelectProvider(type = com.example.provider.UserSqlProvider.class, method = "findByCondition")
    List<User> findByCondition2(Map<String, Object> params);
    
    // ==================== 插入操作 ====================
    
    /**
     * 插入用户并返回自增主键
     */
    @Insert("INSERT INTO user(username, password, age, email, phone) " +
            "VALUES(#{username}, #{password}, #{age}, #{email}, #{phone})")
    @Options(useGeneratedKeys = true, keyProperty = "id", keyColumn = "id")
    int insert(User user);
    
    /**
     * 动态插入（只插入非null字段）
     */
    @InsertProvider(type = com.example.provider.UserSqlProvider.class, method = "insertSelective")
    @Options(useGeneratedKeys = true, keyProperty = "id")
    int insertSelective(Map<String, Object> params);
    
    /**
     * 批量插入
     */
    @Insert({
        "<script>",
        "INSERT INTO user(username, password, age, email) VALUES",
        "<foreach collection='list' item='user' separator=','>",
        "(#{user.username}, #{user.password}, #{user.age}, #{user.email})",
        "</foreach>",
        "</script>"
    })
    int batchInsert(List<User> users);
    
    // ==================== 更新操作 ====================
    
    /**
     * 根据ID更新用户信息
     */
    @Update({
        "<script>",
        "UPDATE user",
        "<set>",
        "<if test='username != null and username != \"\"'>username = #{username},</if>",
        "<if test='password != null and password != \"\"'>password = #{password},</if>",
        "<if test='age != null'>age = #{age},</if>",
        "<if test='email != null and email != \"\"'>email = #{email},</if>",
        "<if test='phone != null and phone != \"\"'>phone = #{phone},</if>",
        "</set>",
        "WHERE id = #{id} AND deleted = 0",
        "</script>"
    })
    int update(User user);
    
    /**
     * 更新密码
     */
    @Update("UPDATE user SET password = #{password} WHERE id = #{id}")
    int updatePassword(@Param("id") Integer id, 
                       @Param("password") String password);
    
    /**
     * 动态更新（使用@UpdateProvider）
     */
    @UpdateProvider(type = com.example.provider.UserSqlProvider.class, method = "updateSelective")
    int updateSelective(Map<String, Object> params);
    
    // ==================== 删除操作 ====================
    
    /**
     * 逻辑删除用户
     */
    @Update("UPDATE user SET deleted = 1 WHERE id = #{id}")
    int delete(Integer id);
    
    /**
     * 批量逻辑删除
     */
    @Update({
        "<script>",
        "UPDATE user SET deleted = 1 WHERE id IN",
        "<foreach collection='ids' item='id' open='(' close=')' separator=','>",
        "#{id}",
        "</foreach>",
        "</script>"
    })
    int batchDelete(@Param("ids") List<Integer> ids);
    
    /**
     * 物理删除（慎用）
     */
    @Delete("DELETE FROM user WHERE id = #{id}")
    int physicalDelete(Integer id);
}

6. SQL Provider类

src/main/java/com/example/provider/UserSqlProvider.java
package com.example.provider;

import org.apache.ibatis.jdbc.SQL;
import java.util.Map;

/**
 * 用户SQL Provider类
 * 用于生成动态SQL
 */
public class UserSqlProvider {
    
    /**
     * 动态查询用户
     */
    public String findByCondition(Map<String, Object> params) {
        return new SQL() {{
            SELECT("*");
            FROM("user");
            WHERE("deleted = 0");
            
            if (params.get("username") != null) {
                AND().WHERE("username LIKE CONCAT('%', #{username}, '%')");
            }
            if (params.get("minAge") != null) {
                AND().WHERE("age >= #{minAge}");
            }
            if (params.get("maxAge") != null) {
                AND().WHERE("age <= #{maxAge}");
            }
            if (params.get("email") != null) {
                AND().WHERE("email = #{email}");
            }
            
            ORDER_BY("id DESC");
        }}.toString();
    }
    
    /**
     * 动态插入用户（只插入非null字段）
     */
    public String insertSelective(Map<String, Object> params) {
        return new SQL() {{
            INSERT_INTO("user");
            
            if (params.get("username") != null) {
                VALUES("username", "#{username}");
            }
            if (params.get("password") != null) {
                VALUES("password", "#{password}");
            }
            if (params.get("age") != null) {
                VALUES("age", "#{age}");
            }
            if (params.get("email") != null) {
                VALUES("email", "#{email}");
            }
            if (params.get("phone") != null) {
                VALUES("phone", "#{phone}");
            }
        }}.toString();
    }
    
    /**
     * 动态更新用户（只更新非null字段）
     */
    public String updateSelective(Map<String, Object> params) {
        return new SQL() {{
            UPDATE("user");
            
            if (params.get("username") != null) {
                SET("username = #{username}");
            }
            if (params.get("password") != null) {
                SET("password = #{password}");
            }
            if (params.get("age") != null) {
                SET("age = #{age}");
            }
            if (params.get("email") != null) {
                SET("email = #{email}");
            }
            if (params.get("phone") != null) {
                SET("phone = #{phone}");
            }
            
            WHERE("id = #{id}");
        }}.toString();
    }
}

7. 测试基类

src/test/java/com/example/test/BaseTest.java
package com.example.test;

import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import org.junit.After;
import org.junit.Before;
import java.io.InputStream;

/**
 * 测试基类
 * 用于初始化SqlSession
 */
public class BaseTest {
    protected SqlSession sqlSession;
    protected SqlSessionFactory sqlSessionFactory;
    
    /**
     * 测试前执行
     * 初始化SqlSession
     */
    @Before
    public void setUp() throws Exception {
        // 1. 读取配置文件
        String resource = "mybatis-config.xml";
        InputStream inputStream = Resources.getResourceAsStream(resource);
        
        // 2. 创建SqlSessionFactory
        sqlSessionFactory = new SqlSessionFactoryBuilder().build(inputStream);
        
        // 3. 创建SqlSession（true表示自动提交）
        sqlSession = sqlSessionFactory.openSession(true);
        
        System.out.println("=== SqlSession初始化完成 ===");
    }
    
    /**
     * 测试后执行
     * 关闭SqlSession
     */
    @After
    public void tearDown() {
        // 关闭SqlSession
        if (sqlSession != null) {
            sqlSession.close();
            System.out.println("=== SqlSession已关闭 ===");
        }
    }
}

10. 注意事项

  1. 修改 mybatis-config.xml 中的数据库连接信息（用户名、密码）
  2. 确保MySQL服务已启动
