三、第1学时：日志框架集成与配置详细步骤

3.1 添加 Log4j 依赖（pom.xml）

操作：在 pom.xml 中添加 Log4j 依赖



    log4j
    log4j
    1.2.17
注意事项：
  • 如果使用 Mybatis 3.5.x，Log4j 1.2.17 版本兼容性好
  • 添加依赖后，右键项目 → Maven → Reload Project

3.2 创建 log4j.properties 配置文件

操作：在 src/main/resources 目录下创建 log4j.properties

# 全局日志级别设置（DEBUG：打印所有级别日志）
log4j.rootLogger=DEBUG, stdout

# 控制台输出配置
log4j.appender.stdout=org.apache.log4j.ConsoleAppender
log4j.appender.stdout.Target=System.out
log4j.appender.stdout.layout=org.apache.log4j.PatternLayout
log4j.appender.stdout.layout.ConversionPattern=%d{yyyy-MM-dd HH:mm:ss} %-5p %c{1}:%L - %m%n

# Mybatis 日志配置（打印 SQL 语句）
log4j.logger.org.apache.ibatis=DEBUG
log4j.logger.java.sql=DEBUG
log4j.logger.java.sql.Statement=DEBUG
log4j.logger.java.sql.PreparedStatement=DEBUG
log4j.logger.java.sql.ResultSet=DEBUG

配置说明：

配置项 说明
log4j.rootLogger 全局日志级别（DEBUG、INFO、WARN、ERROR）
log4j.appender.stdout 输出到控制台
ConversionPattern 日志输出格式（时间、级别、类名、行号、消息）
log4j.logger.org.apache.ibatis Mybatis 日志级别（打印 SQL 语句）

3.3 配置 mybatis-config.xml（启用 Log4j）

操作：在 mybatis-config.xml 中添加 <settings> 配置

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE configuration
        PUBLIC "-//mybatis.org//DTD Config 3.0//EN"
        "http://mybatis.org/dtd/mybatis-3-config.dtd">
<configuration>

    <!-- 引入配置文件 -->
    <properties resource="db.properties"/>

    <!-- 全局设置 -->
    <settings>
        <!-- 使用 Log4j 作为日志实现 -->
        <setting name="logImpl" value="LOG4J"/>
        
        <!-- 开启驼峰命名自动映射 -->
        <setting name="mapUnderscoreToCamelCase" value="true"/>
    </settings>

    <!-- 环境配置 -->
    <environments default="development">
        <environment id="development">
            <transactionManager type="JDBC"/>
            <dataSource type="POOLED">
                <property name="driver" value="${jdbc.driver}"/>
                <property name="url" value="${jdbc.url}"/>
                <property name="username" value="${jdbc.username}"/>
                <property name="password" value="${jdbc.password}"/>
            </dataSource>
        </environment>
    </environments>

    <!-- 注册 Mapper -->
    <mappers>
        <mapper resource="mapper/UserMapper.xml"/>
    </mappers>

</configuration>

操作：运行测试类，查看控制台日志输出

四、第2学时：Junit 单元测试编写详细步骤

4.1 创建标准化测试类

操作：在 src/test/java/com/example 目录下创建 UserMapperTest.java

package com.example;

import com.example.entity.User;
import com.example.mapper.UserMapper;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import org.junit.After;
import org.junit.Before;
import org.junit.Test;

import java.io.IOException;
import java.io.InputStream;
import java.util.List;

/**
 * 用户 Mapper 测试类
 * 测试 UserMapper 接口的所有方法
 */
public class UserMapperTest {

    private InputStream is;
    private SqlSession session;
    private UserMapper userMapper;

    /**
     * 初始化方法（在每个 @Test 方法执行前执行）
     */
    @Before
    public void init() throws IOException {
        System.out.println("初始化方法执行...");
        
        // 1. 加载配置文件
        is = Resources.getResourceAsStream("mybatis-config.xml");

        // 2. 创建 SqlSessionFactory
        SqlSessionFactory factory = new SqlSessionFactoryBuilder().build(is);

        // 3. 创建 SqlSession（true：自动提交事务）
        session = factory.openSession(true);

        // 4. 获取 Mapper 代理对象
        userMapper = session.getMapper(UserMapper.class);
    }

    /**
     * 销毁方法（在每个 @Test 方法执行后执行）
     */
    @After
    public void destroy() throws IOException {
        System.out.println("销毁方法执行...");
        
        // 关闭资源
        if (session != null) {
            session.close();
        }
        if (is != null) {
            is.close();
        }
    }

    /**
     * 测试查询所有用户
     */
    @Test
    public void testFindAll() {
        System.out.println("testFindAll 方法执行...");
        
        // 执行查询
        List users = userMapper.findAll();

        // 遍历结果
        for (User user : users) {
            System.out.println(user);
        }

        // 断言：用户列表不能为空
        assert users != null : "用户列表不能为空";
        assert users.size() > 0 : "用户列表必须有数据";
    }

    /**
     * 测试根据 ID 查询用户
     */
    @Test
    public void testFindById() {
        System.out.println("testFindById 方法执行...");
        
        // 执行查询
        User user = userMapper.findById(1);

        // 打印结果
        System.out.println(user);

        // 断言
        assert user != null : "用户不能为空";
        assert user.getId() == 1 : "用户ID必须是1";
    }
}

4.2 运行测试方法

操作：

  1. 右键 testFindAll() 方法
  2. 选择 Run 'testFindAll()'
  3. 查看控制台输出和日志

运行结果：

初始化方法执行...
testFindAll 方法执行...
2026-07-04 00:20:15 DEBUG JdbcTransaction:137 - Opening JDBC Connection
...
User{id=1, username='张三', ...}
User{id=2, username='李四', ...}

执行顺序：

  1. @Before 方法（init）执行
  2. @Test 方法（testFindAll）执行
  3. @After 方法（destroy）执行

五、实验第1学时：ApiFox 接口测试详细步骤

5.1 下载并安装 ApiFox

  1. 访问 ApiFox 官网：https://www.apifox.cn/
  2. 点击 下载，选择 Windows 版本
  3. 下载完成后，双击安装包，按提示安装
  4. 安装完成后，打开 ApiFox，注册/登录账号

5.2 创建项目与接口

  1. 点击 新建项目，填写项目名称：Mybatis测试项目
  2. 点击 新建接口，填写接口信息：
    • 接口名称：查询所有用户
    • 请求方法：GET
    • 接口路径：http://localhost:8080/user/findAll
  3. 点击 保存

5.3 发送请求，查看响应

  1. 点击 发送 按钮
  2. 查看响应结果（右侧面板）：
    • 状态码：200 OK
    • 响应时间：123 ms
    • 响应数据（JSON格式）
{
  "code": 200,
  "message": "查询成功",
  "data": [
    {
      "id": 1,
      "username": "张三",
      "email": "zhangsan@qq.com"
    },
    {
      "id": 2,
      "username": "李四",
      "email": "lisi@qq.com"
    }
  ]
}

5.4 配置断言（自动验证响应）

  1. 在接口编辑页面，点击 断言 选项卡
  2. 添加断言规则：
    • 状态码等于 200
    • 响应时间小于 1000 ms
    • 返回数据不为空
  3. 保存断言规则
  4. 再次发送请求，查看断言结果

六、实验第2学时：AI 辅助测试开发

6.1 使用通义灵码生成测试代码

  1. 打开测试类 UserMapperTest.java
  2. 在类中输入注释：
// Test method for adding user
  1. 按 Enter，通义灵码会自动生成测试方法
@Test
public void testAddUser() {
    User user = new User();
    user.setUsername("测试用户");
    user.setPassword("123456");
    user.setEmail("test@qq.com");
    
    int rows = userMapper.addUser(user);
    
    assert rows > 0 : "添加失败";
    System.out.println("添加成功，影响行数：" + rows);
}

6.2 使用 AI 优化测试代码

  1. 选中测试代码
  2. 右键 → 通义灵码 → 优化代码
  3. AI 会给出优化建议（如添加注释、简化逻辑、增加断言等）
