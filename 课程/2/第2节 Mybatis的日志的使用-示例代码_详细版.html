📄 2. log4j.properties（Log4j 配置文件）

# 全局日志级别设置
# DEBUG：打印所有级别日志（开发环境）
# INFO：只打印 INFO、WARN、ERROR 级别日志（生产环境推荐）
log4j.rootLogger=DEBUG, stdout, file

# 控制台输出配置
log4j.appender.stdout=org.apache.log4j.ConsoleAppender
log4j.appender.stdout.Target=System.out
log4j.appender.stdout.layout=org.apache.log4j.PatternLayout
# 输出格式：时间 级别 类名:行号 - 消息
log4j.appender.stdout.layout.ConversionPattern=%d{yyyy-MM-dd HH:mm:ss} %-5p %c{1}:%L - %m%n

# 文件输出配置（可选）
log4j.appender.file=org.apache.log4j.RollingFileAppender
log4j.appender.file.File=logs/mybatis.log
log4j.appender.file.MaxFileSize=10MB
log4j.appender.file.MaxBackupIndex=10
log4j.appender.file.layout=org.apache.log4j.PatternLayout
log4j.appender.file.layout.ConversionPattern=%d{yyyy-MM-dd HH:mm:ss} %-5p %c{1}:%L - %m%n

# Mybatis 日志配置（打印 SQL 语句）
log4j.logger.org.apache.ibatis=DEBUG
log4j.logger.java.sql=DEBUG
log4j.logger.java.sql.Statement=DEBUG
log4j.logger.java.sql.PreparedStatement=DEBUG
log4j.logger.java.sql.ResultSet=DEBUG

# 设置特定包的日志级别（可选）
log4j.logger.com.example.mapper=TRACE

配置说明：

  • log4j.rootLogger=DEBUG, stdout, file：全局日志级别为 DEBUG，输出到控制台和文件
  • ConsoleAppender：输出到控制台
  • RollingFileAppender：输出到文件（自动滚动，超过 10MB 创建新文件）
  • ConversionPattern：日志输出格式（时间、级别、类名、行号、消息）

📄 4. UserMapperTest.java（Junit 测试类）

package com.example;

import com.example.entity.User;
import com.example.mapper.UserMapper;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;
import org.junit.After;
import org.junit.Before;
import org.junit.Test;

import java.io.IOException;
import java.io.InputStream;
import java.util.List;

/**
 * 用户 Mapper 测试类
 *
 * 完整测试 UserMapper 接口的所有方法
 */
public class UserMapperTest {

    private InputStream is;
    private SqlSession session;
    private UserMapper userMapper;

    /**
     * 初始化方法（在每个 @Test 方法执行前执行）
     */
    @Before
    public void init() throws IOException {
        System.out.println("========== 初始化开始 ==========");
        
        // 1. 加载配置文件
        is = Resources.getResourceAsStream("mybatis-config.xml");

        // 2. 创建 SqlSessionFactory
        SqlSessionFactory factory = new SqlSessionFactoryBuilder().build(is);

        // 3. 创建 SqlSession（true：自动提交事务）
        session = factory.openSession(true);

        // 4. 获取 Mapper 代理对象
        userMapper = session.getMapper(UserMapper.class);

        System.out.println("========== 初始化完成 ==========");
    }

    /**
     * 销毁方法（在每个 @Test 方法执行后执行）
     */
    @After
    public void destroy() throws IOException {
        System.out.println("========== 销毁开始 ==========");
        
        // 关闭资源
        if (session != null) {
            session.close();
            System.out.println("SqlSession 已关闭");
        }
        if (is != null) {
            is.close();
            System.out.println("InputStream 已关闭");
        }

        System.out.println("========== 销毁完成 ==========");
    }

    /**
     * 测试查询所有用户
     */
    @Test
    public void testFindAll() {
        System.out.println("========== 测试 testFindAll 开始 ==========");
        
        // 执行查询
        List users = userMapper.findAll();

        // 遍历结果
        System.out.println("查询结果：");
        for (User user : users) {
            System.out.println("  " + user);
        }

        // 断言
        assert users != null : "用户列表不能为空";
        assert users.size() > 0 : "用户列表必须有数据";

        System.out.println("========== 测试 testFindAll 完成 ==========");
    }

    /**
     * 测试根据 ID 查询用户
     */
    @Test
    public void testFindById() {
        System.out.println("========== 测试 testFindById 开始 ==========");
        
        // 执行查询
        User user = userMapper.findById(1);

        // 打印结果
        System.out.println("查询结果：" + user);

        // 断言
        assert user != null : "用户不能为空";
        assert user.getId() == 1 : "用户ID必须是1";
        assert user.getUsername() != null : "用户名不能为空";

        System.out.println("========== 测试 testFindById 完成 ==========");
    }

    /**
     * 测试添加用户
     */
    @Test
    public void testAddUser() {
        System.out.println("========== 测试 testAddUser 开始 ==========");
        
        // 创建用户对象
        User user = new User();
        user.setUsername("测试用户");
        user.setPassword("123456");
        user.setEmail("test@qq.com");

        // 执行添加
        int rows = userMapper.addUser(user);

        // 打印结果
        System.out.println("影响行数：" + rows);

        // 断言
        assert rows > 0 : "添加失败";

        System.out.println("========== 测试 testAddUser 完成 ==========");
    }

    /**
     * 测试更新用户
     */
    @Test
    public void testUpdateUser() {
        System.out.println("========== 测试 testUpdateUser 开始 ==========");
        
        // 查询用户
        User user = userMapper.findById(1);

        // 修改信息
        user.setUsername("张三（更新）");
        user.setEmail("zhangsan_updated@qq.com");

        // 执行更新
        int rows = userMapper.updateUser(user);

        // 打印结果
        System.out.println("影响行数：" + rows);

        // 断言
        assert rows > 0 : "更新失败";

        System.out.println("========== 测试 testUpdateUser 完成 ==========");
    }

    /**
     * 测试删除用户
     */
    @Test
    public void testDeleteUser() {
        System.out.println("========== 测试 testDeleteUser 开始 ==========");
        
        // 执行删除（删除 ID=6 的用户）
        int rows = userMapper.deleteUser(6);

        // 打印结果
        System.out.println("影响行数：" + rows);

        // 断言
        assert rows > 0 : "删除失败";

        System.out.println("========== 测试 testDeleteUser 完成 ==========");
    }
}

  2. 响应时间小于 1000 ms
  3. 返回数据不为空
  4. 用户ID等于请求参数ID

📝 常见问题与解决方案

问题 原因 解决方案
控制台没有日志输出 log4j.properties 文件未创建或配置错误 检查 log4j.properties 是否在 src/main/resources 目录下；检查配置是否正确
日志级别设置不生效 log4j.rootLogger 配置错误 确认 log4j.rootLogger=级别, 输出目标；重启应用
Junit 测试报错：ClassNotFoundException 依赖未下载或版本错误 检查 pom.xml，手动执行 mvn clean install
ApiFox 请求失败：Connection refused 后端服务未启动 先启动后端服务（如 Tomcat），再发送请求
